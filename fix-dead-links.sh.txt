#!/bin/bash
# create-missing-en-pages.sh - Erstellt nur fehlende englische Seiten

set -e

VITEPRESS_ROOT="/rep/projects/websites/p2d2-docs"
cd "$VITEPRESS_ROOT"

echo "📝 Erstelle fehlende englische Platzhalter-Seiten..."
echo ""

# Liste der Seiten die in DE existieren, aber in EN fehlen
# Basierend auf .vitepress/config.ts Sidebar

declare -A MISSING_PAGES=(
    # Administrationshandbuch - alle Seiten
    ["en/administrationshandbuch/index.md"]="Administration Handbook"
    ["en/benutzerhandbuch/index.md"]="User Guide"
)

# Erstelle fehlende Dateien
for PAGE in "${!MISSING_PAGES[@]}"; do
    if [ -f "$PAGE" ]; then
        echo "  ℹ️  $PAGE existiert bereits"
        continue
    fi
    
    TITLE="${MISSING_PAGES[$PAGE]}"
    DIR=$(dirname "$PAGE")
    mkdir -p "$DIR"
    
    cat > "$PAGE" << EOFPAGE
# $TITLE

::: warning Page Under Construction
This page is currently under construction. Documentation will be added progressively.
:::

**Status:** In preparation

For more information, see the [Development Strategy](../entwicklungsstrategie/).
EOFPAGE
    
    echo "  ✅ Erstellt: $PAGE"
done

echo ""
echo "✅ Fertig! Teste Build..."

# Test
if npm run docs:build 2>&1 | grep -q "dead link"; then
    echo "⚠️  Es gibt noch Dead Links - manuell prüfen"
    npm run docs:build 2>&1 | grep "dead link"
else
    echo "🎉 Build erfolgreich ohne Dead Links!"
fi
